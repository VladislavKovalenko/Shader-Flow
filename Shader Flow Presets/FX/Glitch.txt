precision mediump float;
uniform vec2  iResolution;
uniform float iTime;
uniform sampler2D uTexture;
uniform int   uHasTexture;
uniform float uAmount; // Общая сила глитча (от 0.0 до 1.0)

// 1. Модуль генерации псевдослучайных чисел (Шум)
float hash(vec2 p) {
    return fract(sin(dot(p, vec2(127.1, 311.7))) * 43758.5453123);
}

void main() {
    vec2 uv = gl_FragCoord.xy / iResolution.xy;
    
    // 2. Модуль расчета искажения (Глитча)
    // Разбиваем экран на горизонтальные полосы, добавляя iTime для динамики
    float blockNoise = hash(vec2(floor(uv.y * 20.0), floor(iTime * 15.0)));
    float lineNoise  = hash(vec2(floor(uv.y * 100.0), floor(iTime * 30.0)));
    
    // Сдвиг происходит только если случайное число превышает порог активности глитча
    float shiftX = 0.0;
    if (blockNoise < uAmount * 0.4) {
        shiftX = (hash(vec2(floor(iTime * 10.0), blockNoise)) - 0.5) * 0.1;
    }
    // Добавляем мелкие и быстрые "игольчатые" полосы
    if (lineNoise < uAmount * 0.2) {
        shiftX += (hash(vec2(floor(iTime * 40.0), lineNoise)) - 0.5) * 0.03;
    }

    // Применяем вычисленный сдвиг к текстурным координатам
    vec2 distortedUV = vec2(fract(uv.x + shiftX), uv.y);

    // 3. Модуль вывода изображения
    if (uHasTexture == 1) {
        // Делаем небольшой разнос каналов (хроматическую аберрацию) только в момент сдвига
        float r = texture2D(uTexture, distortedUV + vec2(shiftX * 0.5, 0.0)).r;
        float g = texture2D(uTexture, distortedUV).g;
        float b = texture2D(uTexture, distortedUV - vec2(shiftX * 0.5, 0.0)).b;
        float al = texture2D(uTexture, distortedUV).a;
        
        gl_FragColor = vec4(r, g, b, al);
    } else {
        // Визуализация глитча без текстуры (процедурная сетка с шумом)
        float stripes = step(0.5, sin(distortedUV.x * 20.0) * cos(distortedUV.y * 20.0));
        vec3 color = vec3(stripes, distortedUV.y, 0.5 + sin(iTime) * 0.5);
        gl_FragColor = vec4(color, 1.0);
    }
}
