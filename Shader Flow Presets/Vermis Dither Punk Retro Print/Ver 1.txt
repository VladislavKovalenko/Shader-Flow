precision mediump float;

vec3 darkColor  = vec3(0.12, 0.20, 0.17); // темно-зеленый
vec3 lightColor = vec3(0.75, 0.88, 0.85); // бирюзовый

float bayer4(vec2 p)
{
    int x = int(mod(p.x, 4.0));
    int y = int(mod(p.y, 4.0));
    int index = x + y * 4;

    if (index == 0)  return 0.0  / 16.0;
    if (index == 1)  return 8.0  / 16.0;
    if (index == 2)  return 2.0  / 16.0;
    if (index == 3)  return 10.0 / 16.0;
    if (index == 4)  return 12.0 / 16.0;
    if (index == 5)  return 4.0  / 16.0;
    if (index == 6)  return 14.0 / 16.0;
    if (index == 7)  return 6.0  / 16.0;
    if (index == 8)  return 3.0  / 16.0;
    if (index == 9)  return 11.0 / 16.0;
    if (index == 10) return 1.0  / 16.0;
    if (index == 11) return 9.0  / 16.0;
    if (index == 12) return 15.0 / 16.0;
    if (index == 13) return 7.0  / 16.0;
    if (index == 14) return 13.0 / 16.0;
    return 5.0 / 16.0; // index == 15
}

float random(vec2 st)
{
    return fract(
        sin(dot(st, vec2(12.9898,78.233)))
        * 43758.5453123
    );
}

void main()
{
    vec2 uv = gl_FragCoord.xy / iResolution.xy;

    if (uHasTexture == 0)
    {
        gl_FragColor = vec4(uv, 0.5, 1.0);
        return;
    }

    vec3 col = texture2D(uTexture, uv).rgb;

    float lum =
        dot(col, vec3(0.299, 0.587, 0.114));

    lum = pow(lum, 0.8);

    float threshold =
        bayer4(gl_FragCoord.xy) * 0.25 - 0.125;

    lum += threshold;

    lum = clamp(lum, 0.0, 1.0);

    float bands = floor(lum * 4.0) / 3.0;

    vec3 duotone =
        mix(darkColor, lightColor, bands);

    float paperNoise =
        random(gl_FragCoord.xy * 0.5);

    duotone +=
        (paperNoise - 0.5) * 0.08;

    vec2 center = uv - 0.5;

    float vignette =
        1.0 - dot(center, center) * 1.2;

    duotone *= vignette;

    gl_FragColor =
        vec4(clamp(duotone, 0.0, 1.0), 1.0);
}
