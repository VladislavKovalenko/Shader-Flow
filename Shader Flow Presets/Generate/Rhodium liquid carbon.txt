precision highp float; // Для Raymarching критически важна highp точность!

uniform float uSpeed;       // Скорость волнения и движения
uniform float uScale;       // Скорость анимации формаций (bounce)
uniform float uGlossiness;  // Яркость свечения объемной небулы/облака

#define time (iTime * speed)
#define res iResolution

// Signed box SDF
float sdBox(vec3 p, vec3 b) {
    vec3 d = abs(p) - b;
    return min(max(d.x, max(d.y, d.z)), 0.0) + length(max(d, 0.0));
}

// 2D Rotation
void pR(inout vec2 p, float a) {
    p = cos(a) * p + sin(a) * vec2(p.y, -p.x);
}

// 3D Noise function (Inigo Quilez)
float noise(vec3 p) {
    vec3 ip = floor(p);
    p -= ip;
    vec3 s = vec3(7.0, 157.0, 113.0);
    vec4 h = vec4(0.0, s.y, s.z, s.y + s.z) + dot(ip, s);
    p = p * p * (3.0 - 2.0 * p);
    h = mix(fract(sin(h) * 43758.5453123), fract(sin(h + s.x) * 43758.5453123), p.x);
    h.xy = mix(h.xz, h.yw, p.y);
    return mix(h.x, h.y, p.z);
}

// Distance Estimator (Карта сцены)
float map(vec3 p, float bounce, float speed) {
    p.z -= 1.0;
    p *= 0.9;
    pR(p.yz, bounce * 1.0 + 0.4 * p.x);
    return sdBox(p + vec3(0.0, sin(1.6 * time), 0.0), vec3(20.0, 0.05, 1.2)) - 0.4 * noise(8.0 * p + 3.0 * bounce);
}

// Вычисление нормалей
vec3 calcNormal(vec3 pos, float bounce, float speed) {
    float eps = 0.001; // Увеличено для стабильности на GPU
    float d = map(pos, bounce, speed);
    return normalize(vec3(
        map(pos + vec3(eps, 0.0, 0.0), bounce, speed) - d,
        map(pos + vec3(0.0, eps, 0.0), bounce, speed) - d,
        map(pos + vec3(0.0, 0.0, eps), bounce, speed) - d
    ));
}

// Raymarching
float castRayx(vec3 ro, vec3 rd, float bounce, float speed) {
    float function_sign = (map(ro, bounce, speed) < 0.0) ? -1.0 : 1.0;
    float precis = 0.001;
    float h = precis * 2.0;
    float t = 0.0;
    for (int i = 0; i < 90; i++) { // Зафиксировано число итераций для стабильности
        if (abs(h) < precis || t > 12.0) break;
        h = function_sign * map(ro + rd * t, bounce, speed);
        t += h;
    }
    return t;
}

// Преломление (Refraction)
float refr(vec3 pos, vec3 lig, vec3 dir, vec3 nor, float angle, out float t2, out vec3 nor2, float bounce, float speed) {
    float h = 0.0;
    t2 = 2.0;
    vec3 dir2 = refract(dir, nor, angle);
    for (int i = 0; i < 40; i++) {
        if (abs(h) > 3.0) break;
        h = map(pos + dir2 * t2, bounce, speed);
        t2 -= h;
    }
    nor2 = calcNormal(pos + dir2 * t2, bounce, speed);
    return (0.5 * clamp(dot(-lig, nor2), 0.0, 1.0) + pow(max(dot(reflect(dir2, nor2), lig), 0.0), 8.0));
}

// Мягкие тени
float softshadow(vec3 ro, vec3 rd, float bounce, float speed) {
    float sh = 1.0;
    float t = 0.02;
    float h = 0.0;
    for (int i = 0; i < 20; i++) {
        if (t > 20.0) break;
        h = map(ro + rd * t, bounce, speed);
        sh = min(sh, 4.0 * h / t);
        t += h;
    }
    return sh;
}

void main() {
    // Нормализация управляющих параметров из диапазона [0.0 .. 1.0]
    float rawSpeed = (uSpeed > 0.0) ? uSpeed : 0.5;
    float speed = mix(0.2, 2.0, rawSpeed);

    float rawScale = (uScale > 0.0) ? uScale : 0.5;
    float bounceMult = mix(5.0, 40.0, rawScale);

    float rawGloss = (uGlossiness > 0.0) ? uGlossiness : 0.5;
    float glowGloss = mix(0.5, 3.0, rawGloss);

    // Рассчет динамики
    float bounce = abs(fract(0.05 * time) - 0.5) * bounceMult;

    // Реакция на мышь или легкое покачивание камеры
    float wobble = (fract(0.1 * (time - 1.0)) >= 0.9) ? fract(-time) * 0.1 * sin(30.0 * time) : 0.0;
    if (iMouse.x > 0.0 || iMouse.y > 0.0) {
        wobble += (iMouse.y / res.y - 0.5) * 0.2;
    }

    // Камера и лучи
    vec3 dir = normalize(vec3(2.0 * gl_FragCoord.xy - res.xy, res.y));
    vec3 org = vec3(0.0, 2.0 * wobble, -3.0);

    vec3 color = vec3(0.0);
    vec3 color2 = vec3(0.0);

    // Трассировка основной поверхности
    float t = castRayx(org, dir, bounce, speed);
    vec3 pos = org + dir * t;
    vec3 nor = calcNormal(pos, bounce, speed);

    // Освещение
    vec3 lig = normalize(vec3(0.2, 6.0, 0.5));
    float depth = clamp((1.0 - 0.09 * t), 0.0, 1.0);

    vec3 pos2 = vec3(0.0);
    vec3 nor2 = vec3(0.0);

    if (t < 12.0) {
        color2 = vec3(max(dot(lig, nor), 0.0) + pow(max(dot(reflect(dir, nor), lig), 0.0), 16.0));
        color2 *= clamp(softshadow(pos, lig, bounce, speed), 0.0, 1.0); // Тень
        
        float t2;
        color2.rgb += refr(pos, lig, dir, nor, 0.9, t2, nor2, bounce, speed) * depth;
        color2 -= clamp(0.1 * t2, 0.0, 1.0); // Потеря интенсивности внутри материала
    }

    // Объемное свечение (Volumetric Glow / Nebula)
    float tmp = 0.0;
    float T = 1.0;
    float intensity = (0.1 * -sin(0.209 * time + 1.0) + 0.05) * glowGloss;

    // Шаги внутри объема (raymarching тумана)
    vec3 rayOrg = org;
    for (int i = 0; i < 64; i++) { // Уменьшено со 128 до 64 для повышения FPS
        float nebula = noise(rayOrg + bounce);
        float density = intensity - map(rayOrg + 0.5 * nor2, bounce, speed) * nebula;
        if (density > 0.0) {
            tmp = density / 64.0;
            T *= 1.0 - tmp * 100.0;
            if (T <= 0.0) break;
        }
        rayOrg += dir * 0.15;
    }

    vec3 basecol = vec3(1.0 / 1.0, 1.0 / 4.0, 1.0 / 16.0); // Оранжево-золотое свечение
    T = clamp(T, 0.0, 1.5);
    color += basecol * exp(4.0 * (0.5 - T) - 0.8);
    color2 *= depth;
    color2 += (1.0 - depth) * noise(6.0 * dir + 0.3 * time) * 0.1; // Легкий туман на фоне

    // Итоговый вывод
    vec3 finalColor = (1.0 * color + 0.8 * color2) * 1.3;

    gl_FragColor = vec4(finalColor, 1.0);
}
